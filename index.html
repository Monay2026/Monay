function loadOnlineUsers() { let sidebarList = document.getElementById('sidebarUsersList'); let mobileList = document.getElementById('onlineUsersListMobile'); let privateListUser = document.getElementById('privateChatsUserList'); let adminList = document.getElementById('adminUserControlList'); let countPC = document.getElementById('onlineCountPC'); let countMobile = document.getElementById('onlineCountMobile'); let usersRef = db.ref(`rooms_data/${currentRoom}/online_users`); // تتبع دخول مستخدم جديد للغرفة (لكافة الزوار، الأعضاء، البوتات والروبوتات) usersRef.off('child_added'); // تفادي تكرار الربط usersRef.on('child_added', (childSnapshot) => { let user = childSnapshot.val(); if(user && user.name) { db.ref(`rooms_data/${currentRoom}/messages`).push({ type: 'system', text: `✨ أهلاً وسهلاً بك في شات مناي (${user.name})، نورت الغرفة!`, time: new Date().toLocaleTimeString([], {hour: '2-digit', minute:'2-digit'}) }); } }); // تتبع خروج مستخدم من الغرفة (لكافة الزوار، الأعضاء، البوتات والروبوتات) usersRef.off('child_removed'); usersRef.on('child_removed', (childSnapshot) => { let user = childSnapshot.val(); if(user && user.name) { db.ref(`rooms_data/${currentRoom}/messages`).push({ type: 'system', text: `👋 رافقتك السلامة (${user.name})، نتمنى أن تعود إلى شات مناي قريباً!`, time: new Date().toLocaleTimeString([], {hour: '2-digit', minute:'2-digit'}) }); } }); // تحديث القوائم البصرية للمتواجدين usersRef.on('value', (snapshot) => { sidebarList.innerHTML = ''; mobileList.innerHTML = ''; privateListUser.innerHTML = ''; adminList.innerHTML = ''; let count = 0; snapshot.forEach((childSnapshot) => { let user = childSnapshot.val(); count++; let genderIcon = user.gender === 'ذكر' ? '👦' : '👧'; let itemHTML = `
${user.name} ${genderIcon}
${user.status}
`; sidebarList.innerHTML += itemHTML; mobileList.innerHTML += itemHTML; privateListUser.innerHTML += itemHTML; if(userRole === "مدير" || userRole === "مشرف") { adminList.innerHTML += `
${user.name} (${user.role})
`; } }); countPC.innerText = count; countMobile.innerText = count; }); }
